# Run from the EITLEM-Kinetics folder. Data root and output root default to the EMULaToR layout:
#   data:    <EMULaToR>/data/processed/datasets/kinetic_dataset/{kcat,km}
#   outputs: <EMULaToR>/outputs/baselines/EITLEM-Kinetics
# Needs a Python env with fair-esm (esm.pretrained.esm1v_t33_650M_UR90S_1), torch_geometric and rdkit.

# 1. All tasks, all default splits, seeds 666 777 888 (the cache is built first)
python emulator_bench/run_split_benchmarks.py \
  --tasks kcat km \
  --seeds 666 777 888 \
  --device cuda:0

# 2. One task only
python emulator_bench/run_split_benchmarks.py --tasks km --device cuda:0

# 3. Build the cache alone
python emulator_bench/cache_embeddings.py --device cuda:0

# 4. Train one explicit split
python emulator_bench/train_single_target_tvt.py \
  --train_path <split_dir>/train.parquet \
  --val_path <split_dir>/val.parquet \
  --test_path <split_dir>/test.parquet \
  --embeddings_dir <output_dir>/embeddings \
  --out_dir <output_dir>/kcat/results/<split_group>/<split_name>/seed_666 \
  --predictor_type kcat \
  --seed 666 \
  --device cuda:0

# 5. Mean and std over seeds
python emulator_bench/aggregate_eitlem_results.py

# 6. Predict from a saved model
python emulator_bench/predict_single_target.py \
  --input_path <split_dir>/test.parquet \
  --embeddings_dir <output_dir>/embeddings \
  --ckpt_path <output_dir>/kcat/results/<split_group>/<split_name>/seed_666/final_model.pth \
  --out_csv <output_dir>/kcat/results/<split_group>/<split_name>/seed_666/test_predictions_from_ckpt.csv \
  --device cuda:0
